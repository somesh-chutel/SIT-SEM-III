<----------------------------JS--------------------------------->


let sizeEl = document.getElementById("size");

let counter = 0;


function onDecrement(){
    counter = counter - 5

    sizeEl.textContent = counter;
}

function onIncrement(){
    counter = counter + 5

    sizeEl.style.fontSize = `${counter}px`;
}
