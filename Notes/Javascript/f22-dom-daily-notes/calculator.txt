let in1El = document.getElementById("in1");

let in2El = document.getElementById("in2");

let opEl = document.getElementById("op");

let resultEl = document.getElementById("result");




function onHavingResult(){
    let val1 = parseInt(in1El.value); //here we are coverting data type into number and then assigning it to variable / and also we are getting the input value
    let val2 = parseInt(in2El.value);
    let operator = opEl.value;
    let result;
    
    console.log(typeof(val1));

    if(operator==="+"){   //we are conferming the oparetor type for arthematic equetion
        result = val1 + val2
    }
    else if(operator==="-"){
        result = val1 - val2

    }
    else if(operator==="/"){
        result = val1 / val2
    }

    resultEl.textContent = result;


}
